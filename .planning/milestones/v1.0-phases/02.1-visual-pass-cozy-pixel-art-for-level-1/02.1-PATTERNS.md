```typescript
// Add facing state field at class level:
private facing: 'down' | 'up' | 'left' | 'right' = 'down';

// AFTER — additive: velocity logic unchanged, anim play added after setVelocity:
update(): void {
    const vx = InputBus.isActive(Action.MOVE_LEFT)  ? -80
             : InputBus.isActive(Action.MOVE_RIGHT) ?  80 : 0;
    const vy = InputBus.isActive(Action.MOVE_UP)    ? -80
             : InputBus.isActive(Action.MOVE_DOWN)  ?  80 : 0;
    this.sprite.setVelocity(vx, vy);

    if (vx === 0 && vy === 0) {
        this.sprite.anims.stop();
        // Idle: show first frame of current facing direction
        const idleFrames: Record<string, number> = { down: 0, up: 2, left: 4, right: 6 };
        this.sprite.setFrame(idleFrames[this.facing]);
    } else {
        if (Math.abs(vx) > Math.abs(vy)) this.facing = vx < 0 ? 'left' : 'right';
        else this.facing = vy < 0 ? 'up' : 'down';
        this.sprite.anims.play(`player_${this.facing}`, true);
        // true = don't restart if already playing this anim (Phaser handles dedup)
    }
}
```

```typescript
// Read first 24 bytes of a PNG to extract width/height from the IHDR chunk.
// PNG spec: bytes 16-19 = width (big-endian uint32), bytes 20-23 = height.
function getPngDimensions(filePath: string): { width: number; height: number } {
    const buf = Buffer.alloc(24);
    const fd = require('fs').openSync(filePath, 'r');
    require('fs').readSync(fd, buf, 0, 24, 0);
    require('fs').closeSync(fd);
    return {
        width:  buf.readUInt32BE(16),
        height: buf.readUInt32BE(20),
    };
}
```

## Critical Constraints (from RESEARCH.md — enforce in all changed files)

1. **Use `tilemap_packed.png` not `tilemap.png`** — packed has 0 margin/spacing; non-packed has 1px between tiles causing render ghosting. `addTilesetImage(..., 16, 16, 0, 0)` — last two params are margin=0, spacing=0.
2. **`-1` is the DECO_DATA empty sentinel, never `0`** — tile 0 is plain grass (valid drawable tile).
3. **Dog patrol corridor** — DECO_DATA rows 13–20, cols 4–5 and 12–17 must all be `-1`. Verified by `art-assets.test.ts`.
4. **No runtime random scatter** — GROUND_DATA and DECO_DATA are pre-computed and committed as static arrays.
5. **No Phaser import in test files** — `tests/art-assets.test.ts` uses only `fs` + static data imports.
6. **61 prior tests must stay green** — changes are in asset files + rendering code only; services/events/save are untouched.
7. **`frameWidth:16, frameHeight:16` for all 16px spritesheets** — must match one frame's pixel dimension exactly (WR-01 lesson).
